# 数据模型定义

## IndexedDB Schema

```typescript
// ── 作品 ──────────────────────────────
interface Work {
  id: string;                // UUID v4
  title: string;             // 作品名
  type: 'novel' | 'short' | 'essay';
  description: string;       // 简介
  coverImage?: string;       // 封面 base64
  wordCount: number;         // 总字数
  chapterCount: number;      // 章节数
  createdAt: number;         // 时间戳
  updatedAt: number;
}

// ── 章节 ──────────────────────────────
interface Chapter {
  id: string;
  workId: string;            // 关联作品
  title: string;
  content: string;           // Markdown 原文
  wordCount: number;
  order: number;             // 排序序号
  status: 'draft' | 'writing' | 'done';
  createdAt: number;
  updatedAt: number;
}

// ── 版本快照 ──────────────────────────
interface ChapterVersion {
  id: string;
  chapterId: string;
  content: string;           // 完整内容快照
  contentHash: string;       // SHA256 前16位
  trigger: 'manual' | 'ai_before_modify' | 'auto_save';
  description: string;       // AI 修改描述
  timestamp: number;
  sequence: number;          // 版本序号
}

// ── 大纲节点 ──────────────────────────
interface OutlineNode {
  id: string;
  workId: string;
  parentId: string | null;   // 树形结构
  type: 'volume' | 'chapter' | 'section' | 'scene';
  title: string;
  summary: string;           // 内容概要
  status: 'planned' | 'writing' | 'done';
  order: number;
  aiScore?: OutlineScore;    // AI 评分
}

interface OutlineScore {
  total: number;             // 0-100
  dimensions: {
    completeness: number;    // 完整度
    pacing: number;          // 节奏
    conflict: number;        // 冲突密度
    characterGrowth: number; // 角色成长
    logic: number;           // 逻辑严谨性
  };
  suggestions: string[];     // AI 改进建议
}

// ── 角色卡片 ──────────────────────────
interface Character {
  id: string;
  workId: string;
  name: string;
  aliases: string[];         // 别名
  role: 'protagonist' | 'antagonist' | 'supporting' | 'minor';
  personality: string;       // 性格描述
  speechStyle: string;       // 说话风格
  background: string;        // 背景故事
  appearance: string;        // 外貌描述
  relationships: CharacterRelation[];
  tags: string[];
  createdAt: number;
  updatedAt: number;
}

interface CharacterRelation {
  targetCharacterId: string;
  relation: string;          // "师徒"、"青梅竹马"、"宿敌"
  intimacy: number;          // -100 ~ 100
  description: string;
}

// ── 世界设定 ──────────────────────────
interface WorldSetting {
  id: string;
  workId: string;
  category: 'geography' | 'history' | 'magic' | 'technology' | 'culture' | 'other';
  title: string;
  content: string;           // Markdown
  tags: string[];
}

// ── AI 对话记录 ────────────────────────
interface AiConversation {
  id: string;
  workId?: string;           // 可选(角色独立对话可不关联作品)
  characterId?: string;      // 可选(角色扮演模式)
  scenario: 'continue' | 'outline_analysis' | 'de_ai' | 'role_chat' | 'reverse_outline';
  messages: AiMessage[];
  createdAt: number;
  updatedAt: number;
}

interface AiMessage {
  role: 'user' | 'assistant' | 'system';
  content: string;
  tokens: number;            // Token 估算
  timestamp: number;
}

// ── 情缘记录 ──────────────────────────
interface BondRecord {
  id: string;
  characterId: string;
  chapterId?: string;        // 触发章节
  event: string;             // 事件描述
  intimacyDelta: number;     // 好感度变化值
  intimacyAfter: number;     // 变化后好感度
  memo: string;              // AI 生成的记忆点
  timestamp: number;
}

// ── 用户设置 ──────────────────────────
interface UserSettings {
  apiKeys: ApiKeyEntry[];    // 加密存储
  modelConfigs: ModelConfig[];
  theme: 'dark' | 'light';
  fontSize: number;          // 编辑器字号
  autoSaveInterval: number;  // 毫秒
  writingStyle: string;      // 默认文风
}

interface ApiKeyEntry {
  provider: string;          // 'deepseek' | 'openai' | 'anthropic' | ...
  key: string;               // 加密后的 Key
  baseUrl: string;
}

interface ModelConfig {
  id: string;
  scenario: string;          // 'continue' | 'outline' | 'de_ai' | 'role_chat'
  model: string;             // 'deepseek-chat' | 'claude-sonnet-4-6' | ...
  temperature: number;       // 0-2
  maxTokens: number;
  provider: string;          // 关联 ApiKeyEntry.provider
}
```

## 本地文件目录布局

```
~/Documents/AI写作引擎/
├── config.json              # 用户设置
├── works/
│   ├── {work-id-1}/
│   │   ├── meta.json        # Work 元数据
│   │   ├── outline.json     # OutlineNode 树
│   │   ├── characters.json  # 角色列表
│   │   ├── settings.json    # WorldSetting 列表
│   │   └── chapters/
│   │       ├── ch-001.json  # Chapter 内容
│   │       ├── ch-001.versions/  # 版本快照
│   │       │   ├── v0001.md
│   │       │   ├── v0002.md
│   │       │   └── ...
│   │       └── ch-002.json
│   └── {work-id-2}/
│       └── ...
└── exports/                 # 导出文件
    ├── {work-title}.epub
    └── {work-title}.txt
```
